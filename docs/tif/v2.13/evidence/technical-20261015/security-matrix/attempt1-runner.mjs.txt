function snapshot() {

    return `${literal(resource.table)},(SELECT COALESCE(jsonb_agg(to_jsonb(t) ORDER BY t.id),'[]'::jsonb) FROM public.${resource.table} t WHERE ${condition})`;
  });
  const rows = JSON.parse(readSql(`SELECT jsonb_build_object(${entries.join(',')})`));
  return { sha256: sha(JSON.stringify(rows)), tables: Object.fromEntries(Object.entries(rows).map(([name, values]) => [name, { rows: values.length, sha256: sha(JSON.stringify(values)), columns: values[0] ? Object.keys(values[0]).sort() : [] }])) };
}
