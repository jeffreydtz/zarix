function adopt(c,response,user){if(response.status!==201||!response.body?.id)return;const id=response.body.id;assert.ok(!c.own.positions.some(x=>x.id===id));const row=position(c,id);assert.equal(row.user_id,user.id);assert.ok(c.own.accounts.some(x=>x.id===row.account_id&&x.userId===user.id));c.own.positions.push({id,user});c.report.fixtures.push({kind:'position',id,userId:user.id});}
