// New binding for the explicitly released intermediate clock retry only.
// No I/O at import; older case results retain their original guard/source snapshots.
import assert from 'node:assert/strict';
import fs from 'node:fs/promises';
import path from 'node:path';
import {spawnSync} from 'node:child_process';
import dotenv from 'dotenv';

export async function bindAccountProfile({root,runtime,profileName,runtimeMode,selected,appState,stateFile,appUrl,apiUrl,config,sourceManifest,digest,sql}){
 const final=profileName==='final-core-acceptance',clock=final?runtimeMode==='clock':true;
 assert.ok(final||profileName==='intermediate-core-reliability-clock-retry');if(final)assert.ok(['clock','normal'].includes(runtimeMode));
 assert.ok((final?['CP2-02','CP2-03','CP2-05','CP2-06','CP2-20','CP3-04','CP3-06','CP3-08','CP3-18']:['CP3-13','CP3-14','CP3-15','CP3-16']).includes(selected));
 assert.equal(appUrl.origin,'http://127.0.0.1:3002');assert.equal(apiUrl.origin,'http://127.0.0.1:58321');
 const profileFile=path.join(root,'docs/tif/v2.13/evidence/technical-20261015/acceptance-final/profiles.json'),profiles=JSON.parse(await fs.readFile(profileFile,'utf8')),profile=profiles[profileName];
 assert.equal(profile.copyRoot,final?'/tmp/zarix-app-acceptance-final-20261015':'/tmp/zarix-app-contract-clock-20261015');assert.equal(profile.coreDirectory,final?'core-acceptance':'core-reliability');assert.equal(profile.fixedRevision,final?null:'836fd18b310a51cced9074b372a80ce5d7e1a870');assert.equal(profile.allowCurrentProductDeltas,!final);if(!final)assert.equal(profile.buildProfile,'intermediate-core-reliability-clock');
 const expectedState=clock?profile.clockState:profile.normalState,geminiPrefix=clock?profile.clockGeminiPrefix:profile.normalGeminiPrefix;
 assert.equal(expectedState,final?(clock?'final-proof-clock-app-state.json':'final-proof-app-state.json'):'contract-clock-retry-app-state.json');assert.equal(stateFile,path.join(runtime,expectedState));assert.equal(geminiPrefix,final?(clock?'final-proof-clock':'final-proof'):'contract-clock-retry');
 const proofFile=path.join(runtime,profile.artifactPrefix+'-build-proof.json'),envFile=path.join(runtime,profile.artifactPrefix+'-app.env'),clockFile=clock?path.join(runtime,profile.clockObservation):null,geminiFile=path.join(runtime,geminiPrefix+'-gemini-unavailable.json');
 for(const file of[stateFile,proofFile,envFile,clockFile,geminiFile].filter(Boolean))assert.equal((await fs.stat(file)).mode&0o077,0);
 const coreFile=path.join(root,'docs/tif/v2.13/evidence/technical-20261015',profile.coreDirectory,'verification.json'),coreBytes=await fs.readFile(coreFile),core=JSON.parse(coreBytes),proofBytes=await fs.readFile(proofFile),proof=JSON.parse(proofBytes);
 assert.equal(core.codeRevision,final?'7c3e8722d8fcdc5d95d205e9a12eb856647e0e06':profile.fixedRevision);assert.equal(core.sourceStable,true);assert.equal(core.sourceWorkingTreeClean,true);
 assert.equal(proof.runtimeProfile,profile.buildProfile||profileName);assert.equal(proof.coreDirectory,profile.coreDirectory);assert.equal(proof.exitCode,0);assert.equal(proof.revision,core.codeRevision);assert.equal(proof.copyRoot,profile.copyRoot);assert.equal(proof.publicSupabaseOrigin,apiUrl.origin);assert.ok(proof.startedAt&&proof.finishedAt);assert.ok(proof.sources.length>0);assert.equal(proof.sourceCount,proof.sources.length);assert.equal(digest(JSON.stringify(proof.sources)),proof.sourceManifestSha256);
 assert.equal(appState.purpose,'account-analysis-proof');assert.equal(appState.runtimeProfile,profileName);assert.equal(appState.coreDirectory,profile.coreDirectory);assert.equal(appState.copyRoot,proof.copyRoot);assert.equal(appState.revision,proof.revision);assert.equal(appState.buildId,proof.buildId);assert.equal(appState.sourceManifestSha256,proof.sourceManifestSha256);assert.equal(appState.privateEnvSha256,proof.privateEnvSha256);assert.equal(appState.clockProfile,clock?'fixed-20261006T120000Z':'real-time');if(clock)assert.equal(appState.fixedTime,'2026-10-06T12:00:00.000Z');assert.equal(appState.guardInstalledBeforeNext,true);assert.equal(appState.adminAlertsDisabled,true);
 const buildIdBytes=await fs.readFile(path.join(proof.copyRoot,'.next/BUILD_ID'));assert.equal(buildIdBytes.toString().trim(),proof.buildId);
 const seen=new Set(),metadata=new Map((proof.manifestOnlyMetadata||[]).map(x=>[x.path,x])),bound=new Map();
 for(const file of proof.sources){assert.ok(!path.isAbsolute(file.path)&&!file.path.split('/').includes('..')&&!seen.has(file.path));seen.add(file.path);bound.set(file.path,file.sha256);assert.equal(digest(await fs.readFile(path.join(proof.copyRoot,file.path))),file.sha256);if(metadata.has(file.path)){assert.equal(file.path,'supabase/.temp/cli-latest');assert.equal(metadata.get(file.path).sha256,file.sha256);}}
 const gitSources=proof.sources.filter(x=>!metadata.has(x.path));assert.ok(gitSources.length>0);const git=spawnSync('git',['cat-file','--batch'],{cwd:root,env:{...process.env,DEVELOPER_DIR:'/Library/Developer/CommandLineTools'},input:gitSources.map(x=>`${proof.revision}:${x.path}`).join('\n')+'\n',maxBuffer:64*1024*1024});assert.equal(git.status,0,'Frozen Git bytes unavailable; raw diagnostics withheld');let cursor=0;
 for(const file of gitSources){const end=git.stdout.indexOf(10,cursor);assert.ok(end>=cursor);const header=git.stdout.subarray(cursor,end).toString().split(' ');assert.equal(header[1],'blob');const size=Number(header[2]);assert.ok(Number.isSafeInteger(size)&&size>=0);cursor=end+1;assert.equal(digest(git.stdout.subarray(cursor,cursor+size)),file.sha256);cursor+=size+1;}assert.equal(cursor,git.stdout.length);
 for(const file of core.sourceManifest)assert.equal(bound.get(file.path),file.sha256);
 const envBytes=await fs.readFile(envFile),env=dotenv.parse(envBytes),nativeFile=path.join(runtime,'proof.env');assert.equal((await fs.stat(nativeFile)).mode&0o077,0);const native=dotenv.parse(await fs.readFile(nativeFile));
 assert.equal(digest(envBytes),proof.privateEnvSha256);assert.equal(env.NEXT_PUBLIC_APP_URL,appUrl.origin);assert.equal(env.NEXT_PUBLIC_SUPABASE_URL,apiUrl.origin);assert.equal(env.NEXT_PUBLIC_SUPABASE_ANON_KEY,config.ZARIX_E2E_ANON_KEY);assert.equal(env.SUPABASE_SERVICE_ROLE_KEY,native.ZARIX_PROOF_SERVICE_ROLE_KEY);assert.equal(env.TELEGRAM_ADMIN_CHAT_ID,'');assert.ok(!Object.entries(env).some(([key,value])=>/GEMINI|GOOGLE.*API.*KEY/i.test(key)&&value));
 const lsof=args=>{const x=spawnSync('/usr/sbin/lsof',args,{encoding:'utf8'});assert.equal(x.status,0,'Exact own listener/cwd absent; raw diagnostics withheld');return x.stdout.split('\n');};
 const listener=lsof(['-nP','-iTCP:3002','-sTCP:LISTEN','-Fpn']);assert.deepEqual([...new Set(listener.filter(x=>x.startsWith('p')).map(x=>Number(x.slice(1))))],[appState.nextPid]);assert.ok(listener.includes('n127.0.0.1:3002'));
 const cwd=lsof(['-a','-p',String(appState.nextPid),'-d','cwd','-Fn']).find(x=>x.startsWith('n')).slice(1);assert.equal(await fs.realpath(cwd),await fs.realpath(proof.copyRoot));
 const prep=JSON.parse(await fs.readFile(path.join(runtime,'proof-private/preparation.json'),'utf8'));assert.equal(prep.status,'PASS_PREPARED');assert.equal(prep.target.apiOrigin,apiUrl.origin);const project=await fs.realpath(path.join(runtime,'proof'));assert.equal(await fs.realpath(prep.target.project),project);assert.equal(digest(await fs.readFile(path.join(project,'supabase/config.toml'))),prep.target.configSha256);
 const stackRoot=path.join(runtime,'home/stacks',prep.target.stackId),stack=JSON.parse(await fs.readFile(path.join(stackRoot,'state.json'),'utf8'));assert.equal(await fs.realpath(stack.identity.projectRoot),project);const db=stack.instances.find(x=>x.creation.service==='database');assert.ok(db);const expectedData=await fs.realpath(path.join(stackRoot,'data',db.id,'data'));assert.equal(await fs.realpath(sql('SHOW data_directory')),expectedData);assert.equal(sql('SELECT current_database()'),'postgres');
 let clockBytes,clockRecord;
 if(clock){ const bytes=await fs.readFile(clockFile),observed=JSON.parse(bytes),clockPreload=path.join(root,'docs/tif/v2.13/evidence/technical-20261015/acceptance-final/clock-preload.mjs');assert.equal(observed.nextPid,appState.nextPid);assert.equal(observed.buildId,proof.buildId);assert.equal(observed.revision,proof.revision);assert.equal(observed.sourceStable,true);assert.equal(observed.fixedTime,appState.fixedTime);assert.equal(observed.observedApplicationTime,appState.fixedTime);assert.equal(observed.preloadSha256,digest(await fs.readFile(clockPreload)));assert.equal(digest(bytes),appState.clockObservationSha256);assert.ok(Array.isArray(observed.command)&&observed.command.includes('3002'));assert.equal(await fs.realpath(observed.copyRoot),await fs.realpath(proof.copyRoot));
clockBytes=bytes;clockRecord=observed;}
 const geminiBytes=await fs.readFile(geminiFile),gemini=JSON.parse(geminiBytes);assert.equal(appState.geminiUnavailableProofFile,geminiFile);assert.equal(digest(geminiBytes),appState.geminiUnavailableProofSha256);assert.equal(gemini.nextPid,appState.nextPid);assert.equal(gemini.buildId,proof.buildId);assert.equal(gemini.mode,'configured-unavailable');assert.equal(gemini.sourceStable,true);assert.equal(gemini.noGeminiEnvironmentCredential,true);assert.equal(gemini.canary.rejectedByInstalledLoopbackGuard,true);assert.equal(gemini.canary.nativeDispatchPreventedByReviewedGuard,true);assert.equal(gemini.instrumentationSha256,digest(await fs.readFile(path.join(root,'docs/tif/v2.13/evidence/technical-20261015/acceptance-final/gemini-unavailable-preload.mjs'))));assert.equal(gemini.journalFile,path.join(runtime,geminiPrefix+'-gemini-attempts.jsonl'));assert.equal((await fs.stat(gemini.journalFile)).mode&0o077,0);
 const currentDeltas=[];for(const file of proof.sources){if(metadata.has(file.path))continue;let current;try{current=digest(await fs.readFile(path.join(root,file.path)));}catch{current=null;}if(current!==file.sha256){if(final&&(/^(?:app|lib|components|supabase|types|public)\//.test(file.path)||/^(?:next\.config\.(?:js|ts)|middleware\.ts|package(?:-lock)?\.json|tsconfig\.json|tailwind\.config\..+)$/.test(file.path)))assert.fail('Final product sources must still match frozen CI5; no unbuilt product acceptance');currentDeltas.push({path:file.path,servedSha256:file.sha256,currentSha256:current,executedByThisBuild:false});}}
 const servedSources=sourceManifest.map(file=>{if(!/^(?:app|lib|components|supabase|types)\//.test(file.path))return file;assert.ok(bound.has(file.path));return{path:file.path,sha256:bound.get(file.path),scope:`Frozen${proof.revision} ${clock?'clock':'normal'} build; current checkout differences disclosed separately`};});
 return {clockFile,geminiFile,servedSources,currentDeltas,clockProof:clock?{path:profile.clockObservation,sha256:digest(clockBytes),fixedTime:clockRecord.fixedTime,observedApplicationTime:clockRecord.observedApplicationTime,preloadSha256:clockRecord.preloadSha256}:undefined,binding:{profile:profileName,runtimeMode:clock?'clock':'normal',revision:proof.revision,copyRoot:proof.copyRoot,buildId:proof.buildId,nextPid:appState.nextPid,appOrigin:appUrl.origin,apiOrigin:apiUrl.origin,coreVerificationSha256:digest(coreBytes),buildProofSha256:digest(proofBytes),physicalSourceCount:proof.sources.length,frozenGitSourceCount:gitSources.length,allPhysicalSourcesVerified:true,allVersionedSourcesMatchFrozenGit:true,privateEnvBoundToExactNativeProject:true,ownedListenerAndCwdVerified:true,dataDirectorySha256:digest(expectedData)},async verifyStable(){for(const file of proof.sources)if(digest(await fs.readFile(path.join(proof.copyRoot,file.path)))!==file.sha256)return false;return digest(await fs.readFile(envFile))===proof.privateEnvSha256&&(!clock||digest(await fs.readFile(clockFile))===digest(clockBytes))&&digest(await fs.readFile(geminiFile))===digest(geminiBytes);}};
}
